### App.OnStart

// --- kolory ---
Set(gblColorBg,          RGBA(244, 244, 245, 1));
Set(gblColorPanel,       RGBA(255, 255, 255, 1));
Set(gblColorAccent,      RGBA(200, 16, 46, 1));
Set(gblColorAccentDark,  RGBA(142, 11, 34, 1));
Set(gblColorAccentSoft,  RGBA(251, 233, 236, 1));
Set(gblColorNav,         RGBA(35, 35, 38, 1));
Set(gblColorText,        RGBA(28, 28, 30, 1));
Set(gblColorMuted,       RGBA(104, 104, 110, 1));
Set(gblColorFaint,       RGBA(134, 134, 141, 1));
Set(gblColorLine,        RGBA(0, 0, 0, 0.09));

// --- kontekst uzytkownika i dat ---
Set(gblUser, User());
Set(gblToday, Today());

// Tydzien Daily Checks zaczyna sie w SOBOTE.
// W sobote Weekday(...) = 1, w piatek = 7, wiec cofamy sie do ostatniej soboty.
Set(gblWeekStart, DateAdd(Today(), -(Weekday(Today(), StartOfWeek.Saturday) - 1), Days));

// --- stan interfejsu ---
Set(gblScreen, "Home");
Set(gblEditMessage, false);
Set(gblNewPage, false);
Set(gblCreating, false);
Set(gblIsManager, false);
Set(gblMessage, Blank());

// PODMIEN na adres swojej witryny, bez ukosnika na koncu
Set(gblSiteUrl, "<ADRES-WITRYNY>")

---


### App.OnStart — wariant z rolami managerow (dopiero gdy beda potrzebne)
// Zastap linie Set(gblIsManager, false) tym blokiem.
// Wymaga konektora Office 365 Groups (standardowy).

Set(gblIsManager,
    !IsEmpty(
        Filter(
            Office365Groups.ListGroupMembers("<ID-GRUPY-MANAGEROW>").value,
            mail = gblUser.Email
        )
    )
)
