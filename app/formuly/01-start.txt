### App.OnStart — wariant z rolami managerow (dopiero gdy beda potrzebne)
// Zastap ostatnia linia Set(gManager, false) tym blokiem.
// Wymaga konektora Office 365 Groups (standardowy) i identyfikatora grupy.

Set(gManager,
    !IsEmpty(
        Filter(
            Office365Groups.ListGroupMembers("<ID-GRUPY-MANAGEROW>").value,
            mail = gJa.Email
        )
    )
)
