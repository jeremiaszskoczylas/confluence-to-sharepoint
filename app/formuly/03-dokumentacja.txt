### scr_Docs.OnVisible
// Strony z biblioteki Site Pages plus wyliczona glebokosc w drzewie.
// Te sama formule wklej tez w ico_Docs_Refresh.OnSelect
// oraz na koncu btn_NewPage_Create.OnSelect (plik 04).

ClearCollect(colPages,
    AddColumns(
        RenameColumns(
            ShowColumns(
                Filter('Site Pages', !IsBlank(TreePath)),
                "ID", "Title", "ParentPageId", "TreePath", "TreeSort", "SortOrder",
                "Service", "Name"
            ),
            "Name", "FileName"
        ),
        "Depth", (Len(TreePath) - Len(Substitute(TreePath, "/", ""))) - 2
    )
);
ClearCollect(colServices,
    SortByColumns(Filter(DT_Services, IsActive), "SortOrder", Ascending)
);
If(IsBlank(gblService), Set(gblService, First(colServices).Code));
ClearCollect(colExpanded, ShowColumns(Filter(colPages, Depth = 0), "ID"));
Set(gblSearch, "")

---

### gal_Docs_Tree.Items
// Wiersz widoczny, gdy jest korzeniem albo gdy WSZYSCY jego przodkowie
// sa rozwinieci. Przy wpisanej frazie drzewo splaszcza sie do wynikow.

SortByColumns(
    Filter(colPages As p,
        p.Service = gblService,
        If(IsBlank(gblSearch),
            p.Depth = 0 ||
            CountRows(
                Filter(colExpanded As e, e.ID <> p.ID && "/" & e.ID & "/" in p.TreePath)
            ) = p.Depth,
            gblSearch in p.Title
        )
    ),
    "TreeSort",
    Ascending
)

---


### ico_Tree_Chevron.X
// wewnatrz galerii gal_Docs_Tree

8 + ThisItem.Depth * 24

### ico_Tree_Chevron.Visible

CountRows(Filter(colPages, ParentPageId = ThisItem.ID)) > 0 && IsBlank(gblSearch)

### ico_Tree_Chevron.Rotation

If(ThisItem.ID in colExpanded.ID, 90, 0)

### ico_Tree_Chevron.OnSelect

If(ThisItem.ID in colExpanded.ID,
    Remove(colExpanded, LookUp(colExpanded, ID = ThisItem.ID)),
    Collect(colExpanded, {ID: ThisItem.ID})
)

---


### lbl_Tree_Page.X

### lbl_Tree_Page.Text

### lbl_Tree_Page.OnSelect

### ico_Docs_Refresh.OnSelect

Refresh('Site Pages');

    AddColumns(
        RenameColumns(
            ShowColumns(
                Filter('Site Pages', !IsBlank(TreePath)),
                "ID", "Title", "ParentPageId", "TreePath", "TreeSort", "SortOrder",
                "Service", "Name"
            ),
            "Name", "FileName"
        ),
        "Depth", (Len(TreePath) - Len(Substitute(TreePath, "/", ""))) - 2
    )
);
ClearCollect(colServices,
    SortByColumns(Filter(DT_Services, IsActive), "SortOrder", Ascending)
);
Notify("Refreshed", NotificationType.Success, 1500)
