### scr_Home.OnVisible
// CALOSC jednym blokiem. Polecenia rozdziela srednik — przy doklejaniu czegokolwiek
// pamietaj o nim, bo jego brak to blad skladni.
// colRosterToday: WSZYSCY z dzisiejszego grafiku (takze urlopy) plus deklaracje
// Daily Checks. Galeria i liczniki korzystaja z tej jednej kolekcji.

ClearCollect(colRosterToday,
    AddColumns(
        Filter(DT_Rota, ShiftDate = gblToday) As r,
        "Checks", LookUp(DT_DailyChecks, WeekStart = gblWeekStart && Title = r.Email)
    )
);
Set(gblMessage,
    First(Sort(Filter(DT_ManagerMessages, ActiveUntil >= gblToday), Created, SortOrder.Descending))
);
UpdateContext(
    {
        loc_BreakStart:
            With({b: LookUp(DT_Breaks, BreakDate = gblToday && Title = gblUser.Email)},
                If(IsBlank(b), "12:00", b.StartTime)
            ),
        loc_BreakCustom: false
    }
)

### lbl_Roster_Status.Text

// Godziny porownujemy jako czas, nie jako tekst: TimeValue("08:00") daje wartosc,
// ktora Power Fx porownuje operatorami bez niespodzianek.
With(
    {
        now: Time(Hour(Now()), Minute(Now()), 0),
        brk: LookUp(DT_Breaks, BreakDate = gblToday && Title = ThisItem.Email)
    },
    If(
        ThisItem.ShiftType.Value <> "Work", ThisItem.ShiftType.Value,
        !IsBlank(brk) && now >= TimeValue(brk.StartTime) && now < TimeValue(brk.EndTime),
            "On break " & brk.StartTime,
        now >= TimeValue(ThisItem.StartTime) && now < TimeValue(ThisItem.EndTime), "Working",
        "Off shift"
    )
)

---


### lbl_Roster_Status.Color

With(
    {
        now: Time(Hour(Now()), Minute(Now()), 0),
        brk: LookUp(DT_Breaks, BreakDate = gblToday && Title = ThisItem.Email)
    },
    If(
        ThisItem.ShiftType.Value <> "Work", gblColorMuted,
        !IsBlank(brk) && now >= TimeValue(brk.StartTime) && now < TimeValue(brk.EndTime), gblColorAccent,
        now >= TimeValue(ThisItem.StartTime) && now < TimeValue(ThisItem.EndTime), gblColorText,
        gblColorMuted
    )
)

### lbl_Break_End.Text
// Koniec przerwy = poczatek + godzina, takze dla wpisu wlasnego

    "until " &
    Text(Mod(Value(Left(s, 2)) + 1, 24), "00") & ":" & Right(s, 2)
)

### btn_Break_Save.Text

"Save break"

### btn_Break_Save.DisplayMode

If(IsMatch(loc_BreakStart, "^([01][0-9]|2[0-3]):[0-5][0-9]$"),
    DisplayMode.Edit,
    DisplayMode.Disabled
)

### btn_Break_Save.OnSelect

With(
    {
        existing: LookUp(DT_Breaks, BreakDate = gblToday && Title = gblUser.Email),
        startAt: loc_BreakStart,
        endAt: Text(Mod(Value(Left(loc_BreakStart, 2)) + 1, 24), "00") & ":" & Right(loc_BreakStart, 2)
    },
    If(IsBlank(existing),
        Patch(DT_Breaks, Defaults(DT_Breaks),
            {
                Title: gblUser.Email,
                Person: {
                    '@odata.type': "#Microsoft.Azure.Connectors.SharePoint.SPListExpandedUser",
                    Claims: "i:0#.f|membership|" & gblUser.Email,
                    DisplayName: gblUser.FullName,
                    Email: gblUser.Email
                },
                BreakDate: gblToday,
                StartTime: startAt,
                EndTime: endAt
            }
        ),
        Patch(DT_Breaks, existing, {StartTime: startAt, EndTime: endAt})
    )
);
Notify("Break saved", NotificationType.Success, 2000)

### ico_Home_Refresh.OnSelect

Refresh(DT_Breaks);
Refresh(DT_DailyChecks);
Refresh(DT_ManagerMessages);

Set(gblWeekStart, DateAdd(Today(), -(Weekday(Today(), StartOfWeek.Saturday) - 1), Days));
Set(gblMessage,
    First(Sort(Filter(DT_ManagerMessages, ActiveUntil >= gblToday), Created, SortOrder.Descending))
);
ClearCollect(colRosterToday,
    AddColumns(
        Filter(DT_Rota, ShiftDate = gblToday) As r,
        "Checks", LookUp(DT_DailyChecks, WeekStart = gblWeekStart && Title = r.Email)
