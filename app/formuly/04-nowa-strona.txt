// --------------------------------------------------------------------------
// UTWORZENIE STRONY

### btnNowaUtworz.DisplayMode

    DisplayMode.Disabled,
    DisplayMode.Edit
)

---


### btnNowaUtworz.OnSelect
// Kolejnosc: utworz strone -> przelicz drzewo -> odswiez kolekcje -> otworz strone.
// Przeliczenie drzewa trwa kilkanascie sekund, stad wskaznik gTworze.

Set(gTworze, true);
Set(gWynik,
    NowaStrona.Run(
        Trim(txtNowyTytul.Text),
        Text(gRodzic),
        drpNowySerwis.Selected.Kod,
        Coalesce(drpNowaKategoria.Selected.Value, "")
    )
);
OdswiezDrzewo.Run();
Refresh('Site Pages');
ClearCollect(colStrony,
    AddColumns(
        RenameColumns(
            ShowColumns(
                Filter('Site Pages', !IsBlank(TreePath)),
                "ID", "Title", "ParentPageId", "TreePath", "TreeSort", "SortOrder",
                "Service", "Name"
            ),
            "Name", "Plik"
        ),
        "Poziom", (Len(TreePath) - Len(Substitute(TreePath, "/", ""))) - 2
    )
);
Set(gTworze, false);
Set(gNowaStrona, false);
Notify("Strona utworzona", NotificationType.Success, 3000);
Launch(gWitrynaUrl & Mid(gWynik.adres, Find("/SitePages/", gWynik.adres)), {}, LaunchTarget.New)

---
