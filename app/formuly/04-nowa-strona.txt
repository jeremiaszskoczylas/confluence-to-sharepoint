### btn_NewPage_Create.Text

"Create page"

### btn_NewPage_Create.DisplayMode

If(IsBlank(gblParentId) || IsBlank(Trim(txt_NewPage_Title.Text)),
    DisplayMode.Disabled,
    DisplayMode.Edit
)

### btn_NewPage_Create.OnSelect
// Kolejnosc: utworz strone -> przelicz drzewo -> odswiez kolekcje -> otworz strone.

Set(gblCreating, true);
Set(gblCreateResult,
    NowaStrona.Run(
        Trim(txt_NewPage_Title.Text),
        Text(gblParentId),
        LookUp(colServices, Title = drp_NewPage_Service.Selected.Title).Code,
        Coalesce(drp_NewPage_Category.Selected.Result, "")
    )
);
OdswiezDrzewo.Run();
Refresh('Site Pages');
ClearCollect(colPages,
    AddColumns(
        RenameColumns(
            ShowColumns(
                Filter('Site Pages', TreeSort <> ""),
                "ID", "Title", "ParentPageId", "TreePath", "TreeSort", "SortOrder",
                "Service", "{FilenameWithExtension}"
            ),
            "{FilenameWithExtension}", "FileName"
        ),
        "Depth", (Len(TreePath) - Len(Substitute(TreePath, "/", ""))) - 2
    )
);
Set(gblCreating, false);
Set(gblNewPage, false);
Notify("Page created", NotificationType.Success, 3000);
// Gdy flow nie zwroci adresu, nie probujemy otwierac pustego odnosnika.
With({a: gblCreateResult.adres},
    If(!IsBlank(a),
        Launch(gblSiteUrl & Mid(a, Find("/SitePages/", a)), {}, LaunchTarget.New)
    )
)
